'---------------Связи между таблицами -------------'
pk, fk

pk- primari key - первичный ключ, ограниченние, которое накладывается на поле , которое будет использоваться в связи 
(создает ,btree для быстрого поиска)

fk-foreign key - внешний ключ, ограничение, которое накладывается на поле, которое ссылается на pkв другой таблицы

'---------one to one -связь-----'
CREATE TABLE author (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE book(
    id SERIAL PRIMARY KEY,
    title  VARCHAR(100),

    author_id   INT UNIQUE,
    CONSTRAINT fk_book_aothor FOREIGN KEY(author_id)
    REFERENCES author (id)
);

'----------one to many------------'
CREATE TABLE author (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)

'--------many to many -----'
CREATE TABLE author (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE book(
    id SERIAL PRIMARY KEY,
    title  VARCHAR(100)
);

CREATE TABLE author_book(
    author_id INT,
    book_id INT,

    CONSTRAINT fk_author FOREIGN KEY 
    (author_id) REFERENCES author(id),

    CONSTRAINT fk_book FOREIGN KEY
    (book_id) REFERENCES book (id)
)

'----------------види Связи--------------------'
#One to one - один к одному
один человек - один ID 
один человек - одно лицо

#One to many -один ко многим
один блогер - много постов
один куратор- много студенты
один продукт- много коментариев

#Many to many - многа ко многим
один ментр - много студентов, один студент - много менторов
один разработчик - много проектов, один проект - много разработчик
один банк - много клинтов, один клиент - много банков
